def digest(data): return hashlib.sha256(data).hexdigest()
def write(name, data):
    with (out/name).open('xb') as f: f.write(data)
def write_json(name, obj): write(name, (json.dumps(obj, indent=2)+'\n').encode())
def compress(name, source):
    data = source.read_bytes()
    encoded = gzip.compress(data, mtime=0)
    assert gzip.decompress(encoded) == data
    write(name, encoded)
    return {'source':str(source), 'retained':name, 'encoding':'gzip (lossless)',
            'decoded_bytes':len(data), 'decoded_sha256':digest(data),
            'stored_bytes':len(encoded), 'stored_sha256':digest(encoded)}
